# Hidden deployment event: wait for CPS/war access to refresh, then create the
# temporary JAP expeditionary force in Guangzhou. Retry only while the war lasts.
country_event = {
	id = DOP_GNG_flow.16
	hidden = yes
	is_triggered_only = yes
	trigger = {
		NOT = { has_country_flag = DOP_JAP_southwest_reinforcements_deployed }
		GNG = {
			has_country_flag = DOP_GNG_flow_southwest_war_active
			has_country_flag = DOP_GNG_flow_japanese_reinforcements_arrived
			has_capitulated = no
		}
	}
	immediate = {
		if = {
			limit = {
				has_war_with = YUN
				592 = {
					OR = {
						is_controlled_by = GNG
						is_controlled_by = JAP
					}
				}
			}
			load_oob = "DOP_JAP_southwest_reinforcements"
			592 = {
				create_unit = {
					division = "division_template = \"DOP Nanshin Sensha Shidan\" start_experience_factor = 0.85 start_equipment_factor = 1.0"
					owner = JAP
					count = 12
				}
			}
			set_country_flag = DOP_JAP_southwest_reinforcements_deployed
			GNG = { country_event = { id = DOP_GNG_flow.4 hours = 1 } }
			log = "[GetDateText]: [Root.GetName]: deployed twelve DOP JAP armoured divisions in Guangzhou"
		}
		else_if = {
			limit = {
				GNG = {
					has_country_flag = DOP_GNG_flow_southwest_war_active
					has_capitulated = no
				}
			}
			country_event = { id = DOP_GNG_flow.16 hours = 12 }
			log = "[GetDateText]: [Root.GetName]: Guangzhou unavailable; retrying DOP JAP deployment in twelve hours"
		}
	}
}
