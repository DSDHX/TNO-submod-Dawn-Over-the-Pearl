scripted_gui = {
	DOP_construction_base_container = {
		window_name = "DOP_construction_base_container"
		context_type = player_context
		parent_window_name = powerbalanceview
		visible = {
			check_variable = { TNO_BoP_SelectedTab = token:BoP_Tab_DOPConstruction }
		}
		ai_enabled = { always = no }

		dynamic_lists = {
			construction_region_grid = {
				array = DOP_construction_region_ids
				index = DOP_construction_region_list_index
				value = DOP_construction_region_id
				change_scope = no
				entry_container = DOP_construction_region_entry
			}
		}
		dynamic_lists = {
			construction_project_grid = {
				array = DOP_construction_visible_projects
				index = DOP_construction_project_list_index
				value = DOP_construction_project_id
				change_scope = no
				entry_container = DOP_construction_project_entry
			}
		}
		dynamic_lists = {
			DOP_construction_funding_set_buttons = {
				array = DOP_construction_slider_set_indices
				index = DOP_construction_funding_set_list_index
				value = DOP_construction_funding_set_index
				change_scope = no
				entry_container = DOP_construction_funding_set_button_entry
			}
		}
		dynamic_lists = {
			DOP_construction_manpower_set_buttons = {
				array = DOP_construction_slider_set_indices
				index = DOP_construction_manpower_set_list_index
				value = DOP_construction_manpower_set_index
				change_scope = no
				entry_container = DOP_construction_manpower_set_button_entry
			}
		}

		effects = {
			construction_region_entry_button_click = {
				set_variable = { DOP_construction_selected_region = DOP_construction_region_id }
				DOP_construction_rebuild_visible_projects = yes
				log = "[GetDateText]: [Root.GetName]: DOP construction region selected [?DOP_construction_selected_region]"
			}
			construction_project_entry_button_click = {
				set_variable = { DOP_construction_selected = DOP_construction_project_id }
				DOP_construction_sync_selected = yes
				log = "[GetDateText]: [Root.GetName]: DOP construction project selected [?DOP_construction_selected]"
			}
			DOP_construction_funding_set_button_click = {
				set_temp_variable = { DOP_construction_target_project = DOP_construction_selected }
				set_temp_variable = { DOP_construction_value = DOP_construction_funding_set_index }
				multiply_temp_variable = { DOP_construction_value = 1.5 }
				add_to_temp_variable = { DOP_construction_value = 10 }
				DOP_construction_set_funding = yes
			}
			DOP_construction_manpower_set_button_click = {
				set_temp_variable = { DOP_construction_target_project = DOP_construction_selected }
				set_temp_variable = { DOP_construction_value = DOP_construction_manpower_set_index }
				multiply_temp_variable = { DOP_construction_value = 1.5 }
				add_to_temp_variable = { DOP_construction_value = 10 }
				DOP_construction_set_manpower = yes
			}
			construction_funding_decrease_click = { DOP_construction_selected_funding_decrease = yes }
			construction_funding_increase_click = { DOP_construction_selected_funding_increase = yes }
			construction_funding_decrease_control_click = { DOP_construction_selected_funding_minimum = yes }
			construction_funding_increase_control_click = { DOP_construction_selected_funding_maximum = yes }
			construction_manpower_decrease_click = { DOP_construction_selected_manpower_decrease = yes }
			construction_manpower_increase_click = { DOP_construction_selected_manpower_increase = yes }
			construction_manpower_decrease_control_click = { DOP_construction_selected_manpower_minimum = yes }
			construction_manpower_increase_control_click = { DOP_construction_selected_manpower_maximum = yes }
		}

		properties = {
			construction_main_progress = {
				frame = DOP_construction_selected_percent
			}
			construction_funding_dot = {
				x = DOP_construction_funding_thumb_x
			}
			construction_manpower_dot = {
				x = DOP_construction_manpower_thumb_x
			}
			construction_project_entry_progress = {
				frame = DOP_construction_percent^DOP_construction_project_id
			}
			construction_project_image = {
				image = "GFX_[?DOP_construction_selected_image.GetTokenKey]"
			}
		}
	}
}
