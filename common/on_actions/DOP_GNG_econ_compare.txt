# DOP: 珠江奇迹经济图 - 每月滚动镜像数据
# TNO 的 on_monthly_GNG 实际未被任何 fire_on_actions 调用，因此这里直接挂在
# 引擎内置的 on_monthly 上，并用 tag 守卫只对 GNG 生效。
on_actions = {
	on_monthly = {
		effect = {
			if = {
				limit = {
					tag = GNG
					NOT = {
						has_country_flag = GNG_disable_all_mechanics_IJA
					}
				}
				DOP_GNG_econ_compare_monthly = yes
			}
		}
	}
}
