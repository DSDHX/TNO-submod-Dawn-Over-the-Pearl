on_actions = {
    on_daily = {
		effect = {
			# Existing saves may have activated the construction/GSA host first.
			# Repair on the next daily tick without resetting tabs or SCW progress.
			if = {
				limit = {
					original_tag = GNG
					check_variable = { TNO_BoP_Tabs^num > 0 }
					OR = {
						NOT = { has_power_balance = { id = BoP_Tab_DOPSiliconCW_GNG } }
						has_power_balance = { id = BoP_Tab_DOPConstruction_GNG }
						has_power_balance = { id = BoP_Tab_DOPScienceAcademy_GNG }
					}
				}
				DOP_BoP_ensure_shared_host = yes
			}
			if = {
				limit = {
					has_country_flag = DOP_construction_initialized
					NOT = { has_country_flag = DOP_construction_dynamic_v3_initialized }
				}
				DOP_construction_migrate_dynamic_v3 = yes
			}
			if = {
				limit = {
					has_country_flag = DOP_construction_initialized
					NOT = { has_country_flag = DOP_construction_dynamic_v4_initialized }
				}
				DOP_construction_migrate_dynamic_v4 = yes
			}
			if = {
				limit = {
					has_country_flag = DOP_construction_initialized
					NOT = { has_country_flag = DOP_construction_dynamic_v5_initialized }
				}
				DOP_construction_migrate_dynamic_v5 = yes
			}
			if = {
				limit = {
					has_country_flag = DOP_construction_initialized
					NOT = { has_country_flag = DOP_construction_dynamic_v6_initialized }
				}
				DOP_construction_migrate_dynamic_v6 = yes
			}
			if = {
				limit = {
					has_country_flag = DOP_construction_initialized
					NOT = { has_country_flag = DOP_construction_dynamic_v7_initialized }
				}
				DOP_construction_migrate_dynamic_v7 = yes
			}
			if = {
				limit = {
					has_country_flag = DOP_construction_initialized
					NOT = { has_country_flag = DOP_construction_dynamic_v8_initialized }
				}
				DOP_construction_migrate_dynamic_v8 = yes
			}
			if = {
				limit = { has_country_flag = DOP_construction_initialized }
				add_to_variable = { DOP_construction_weekly_counter = 1 }
				if = {
					limit = {
						check_variable = {
							var = DOP_construction_weekly_counter
							value = 7
							compare = greater_than_or_equals
						}
					}
					subtract_from_variable = { DOP_construction_weekly_counter = 7 }
					DOP_construction_weekly_tick = yes
				}
				DOP_construction_process_completion_queue = yes
			}
			if = {
				limit = {
					check_variable = { GNG_GSA_research_on_now > 0 }
				}
				DOP_GSA_refresh_research_rate = yes
				add_to_variable = { GNG_GSA_research_progress = GNG_GSA_research_speed_percent }
				if = {
					limit = {
						check_variable = {
							var = GNG_GSA_research_progress
							value = 100
							compare = greater_than_or_equals
						}
					}
					GSA_research_complete_effects = yes
					log = "GNG:Research complete for [?GNG_GSA_research_on_now]"
					set_variable = { GNG_GSA_research_on_now = 0 }
					set_variable = { GNG_GSA_research_progress = 0 }
					DOP_GSA_refresh_research_rate = yes
				}
			}
		}
	}
}
