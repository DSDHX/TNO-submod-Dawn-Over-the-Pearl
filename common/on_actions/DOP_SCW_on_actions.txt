on_actions = {
	on_daily = {
		effect = {
			# Persist the display fallback for old saves without resetting valid selections.
			if = {
				limit = {
					has_country_flag = DOP_SCW_enabled
					NOT = { DOP_SCW_chain_selection_valid = yes }
				}
				DOP_SCW_ensure_chain_selection = yes
			}
			if = {
				limit = {
					original_tag = GNG
					has_country_flag = DOP_SCW_enabled
					NOT = { has_country_flag = DOP_SCW_decisions_unlocked }
				}
				DOP_SCW_activate_decision_system = yes
			}
		}
	}
	on_yearly = {
		effect = {
			if = {
				limit = {
					original_tag = GNG
					has_country_flag = DOP_SCW_initialized
					has_country_flag = DOP_SCW_decisions_unlocked
				}
				hidden_effect = { DOP_SCW_apply_opponent_yearly_growth = yes }
			}
		}
	}
}
