DOP_construction_clear_completion_flags = {
	for_each_loop = {
		array = DOP_construction_project_ids
		value = DOP_construction_project_id
		set_temp_variable = { DOP_construction_project_token = DOP_construction_project_tokens^DOP_construction_project_id }
		meta_effect = {
			text = { clr_country_flag = [PROJECT]_completed }
			PROJECT = "[?DOP_construction_project_token.GetTokenKey]"
		}
	}
}

DOP_construction_migrate_legacy_variables = {
	set_temp_variable = { DOP_construction_legacy_base_speed = DOP_construction_base_speed }
	set_temp_variable = { DOP_construction_legacy_selected = DOP_construction_selected }
	DOP_construction_register_all_projects = yes
	set_variable = { DOP_construction_base_speed = DOP_construction_legacy_base_speed }
	divide_variable = { DOP_construction_base_speed = 4 }
	set_variable = { DOP_construction_selected = DOP_construction_legacy_selected }
	clamp_variable = { var = DOP_construction_selected min = 1 max = DOP_construction_project_count }

	for_each_loop = {
		array = DOP_construction_project_ids
		value = DOP_construction_project_id
		set_temp_variable = { DOP_construction_project_token = DOP_construction_project_tokens^DOP_construction_project_id }
		meta_effect = {
			text = {
				set_variable = { DOP_construction_progress^[PROJECT_ID] = [PROJECT]_progress }
				set_variable = { DOP_construction_funding^[PROJECT_ID] = [PROJECT]_funding }
				set_variable = { DOP_construction_manpower^[PROJECT_ID] = [PROJECT]_manpower }
				set_variable = { DOP_construction_funding_min^[PROJECT_ID] = [PROJECT]_funding_min }
				set_variable = { DOP_construction_funding_max^[PROJECT_ID] = [PROJECT]_funding_max }
				set_variable = { DOP_construction_manpower_min^[PROJECT_ID] = [PROJECT]_manpower_min }
				set_variable = { DOP_construction_manpower_max^[PROJECT_ID] = [PROJECT]_manpower_max }
				set_variable = { DOP_construction_speed_add^[PROJECT_ID] = [PROJECT]_speed_add }
				set_variable = { DOP_construction_speed_factor^[PROJECT_ID] = [PROJECT]_speed_factor }
				if = {
					limit = { has_country_flag = [PROJECT]_completed }
					set_variable = { DOP_construction_completed^[PROJECT_ID] = 1 }
				}
			}
			PROJECT = "[?DOP_construction_project_token.GetTokenKey]"
			PROJECT_ID = "[?DOP_construction_project_id|.0]"
		}
	}

	set_variable = { DOP_construction_selected_region = DOP_construction_project_region^DOP_construction_selected }
	set_country_flag = DOP_construction_initialized
	set_country_flag = DOP_construction_dynamic_v2_initialized

	DOP_construction_rebuild_visible_projects = yes
}
