# value variable documented per effect.
DOP_construction_finish_external_change = {
	set_temp_variable = { DOP_construction_project_id = DOP_construction_target_project }
	DOP_construction_recalculate_project = yes
	DOP_construction_complete_project_if_needed = yes
	if = {
		limit = { check_variable = { DOP_construction_selected = DOP_construction_target_project } }
		DOP_construction_sync_selected = yes
	}
	else = { add_to_variable = { DOP_construction_gui_dirty = 1 } }
}
