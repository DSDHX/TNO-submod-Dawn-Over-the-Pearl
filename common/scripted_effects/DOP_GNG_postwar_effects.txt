DOP_GNG_update_stage_integrity = {
	remove_ideas = DOP_GNG_integrity_waltz
	remove_ideas = DOP_GNG_integrity_chorus
	remove_ideas = DOP_GNG_integrity_dissonance
	remove_ideas = DOP_GNG_integrity_chaos
	if = {
		limit = { check_variable = { DOP_GNG_stage_integrity >= 75 } }
		add_ideas = DOP_GNG_integrity_waltz
	}
	else_if = {
		limit = { check_variable = { DOP_GNG_stage_integrity >= 50 } }
		add_ideas = DOP_GNG_integrity_chorus
	}
	else_if = {
		limit = { check_variable = { DOP_GNG_stage_integrity >= 25 } }
		add_ideas = DOP_GNG_integrity_dissonance
	}
	else = {
		add_ideas = DOP_GNG_integrity_chaos
	}
}
