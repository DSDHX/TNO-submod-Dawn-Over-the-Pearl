DOP_GNG_refresh_fiscal_from_debt = {
	if = {
		limit = { check_variable = { debt_to_GDP_ratio < 0.25 } }
		set_temp_variable = { GNG_fiscal_status_var_temp = 0 }
	}
	else_if = {
		limit = { check_variable = { debt_to_GDP_ratio < 0.50 } }
		set_temp_variable = { GNG_fiscal_status_var_temp = 1 }
	}
	else_if = {
		limit = { check_variable = { debt_to_GDP_ratio < 0.75 } }
		set_temp_variable = { GNG_fiscal_status_var_temp = 2 }
	}
	else_if = {
		limit = { check_variable = { debt_to_GDP_ratio < 1.00 } }
		set_temp_variable = { GNG_fiscal_status_var_temp = 3 }
	}
	else = {
		set_temp_variable = { GNG_fiscal_status_var_temp = 4 }
	}
	GNG_update_fiscal = yes
}

DOP_GNG_apply_southwest_war_state_modifiers = {
	DOP_GNG_remove_southwest_war_state_modifiers = yes
	GNG = {
		every_owned_state = {
			set_state_flag = DOP_GNG_southwest_defence_modifier_state
			if = {
				limit = { GNG_is_pearl_river_delta_state = yes }
				add_dynamic_modifier = { modifier = DOP_GNG_prd_wartime_defence }
			}
			else = {
				add_dynamic_modifier = { modifier = DOP_GNG_outer_wartime_defence }
			}
		}
	}

	# Lock both sides of the real YUN-CHI land frontier. The GNG-YUN front is
	# deliberately excluded so Long Shengwu's army can press into Guangdong.
	every_state = {
		limit = {
			OR = {
				AND = {
					is_controlled_by = YUN
					any_neighbor_state = { is_controlled_by = CHI }
				}
				AND = {
					is_controlled_by = CHI
					any_neighbor_state = { is_controlled_by = YUN }
				}
			}
		}
		set_state_flag = DOP_GNG_southwest_stalemate_state
		add_dynamic_modifier = { modifier = DOP_GNG_southwest_stalemate }
	}
}

DOP_GNG_begin_southwest_crisis = {
	DOP_GNG_block_native_western_insurrection_responses = yes
	if = {
		limit = { NOT = { has_country_flag = DOP_GNG_flow_southwest_crisis_started } }
		# DOP supplies its own date and declaration sequence. This only prepares
		# the YUN government and reusable TNO mechanics before that sequence.
		YUN = { DOP_YUN_prepare_southwest_war = yes }
		set_country_flag = DOP_GNG_flow_southwest_crisis_started
		set_variable = { DOP_GNG_content_stage = 2 }
		load_focus_tree = beating_drums_tree

		# Guangxi and all three Hainan states join the Yunnan revolt.
		if = {
			limit = { country_exists = GUX }
			YUN = { annex_country = { target = GUX transfer_troops = yes } }
		}
		YUN = {
			transfer_state = 591
			transfer_state = 2474
			transfer_state = 2475
		}

		# Yunnan begins as CHI's subject in TNO; the revolt must free it before
		# either declaration of war can be legal.
		CHI = {
			set_autonomy = {
				target = YUN
				autonomy_state = autonomy_free
			}
		}

		# Mirror the already tested DOP debug scope: Japan is the CPS leader.
		JAP = {
			remove_from_faction = CHI
			remove_from_faction = GNG
			remove_from_faction = YUN
		}

		DOP_GNG_apply_southwest_war_state_modifiers = yes
		YUN = {
			if = {
				limit = { NOT = { has_war_with = CHI } }
				declare_war_on = { target = CHI type = annex_everything }
			}
		}
		country_event = { id = DOP_GNG_flow.10 hours = 1 }
		country_event = { id = DOP_GNG_flow.2 days = 14 }
		log = "[GetDateText]: [Root.GetName]: DOP southwest crisis started; YUN absorbed GUX and Hainan"
	}
}

DOP_GNG_enter_southwest_war = {
	if = {
		limit = { NOT = { has_country_flag = DOP_GNG_flow_southwest_war_active } }
		set_country_flag = DOP_GNG_flow_southwest_war_active
		set_variable = { DOP_GNG_content_stage = 2 }
		YUN = {
			if = {
				limit = { NOT = { has_war_with = GNG } }
				declare_war_on = { target = GNG type = annex_everything }
			}
		}

		# The existing GNG 104th armoured formation is removed without refunding
		# equipment. The later relief force uses JAP's own Sensha Shidan template.
		delete_units = {
			division_template = "Sensha Shidan - Armored Division"
			disband = no
		}

		DOP_GNG_set_status_first_crisis = yes
		DOP_GNG_set_security_war = yes
		set_temp_variable = { GNG_corruption_status_var_temp = 3 }
		GNG_update_corruption = yes
		DOP_GNG_refresh_fiscal_from_debt = yes
		add_ideas = DOP_GNG_wartime_triangle
		add_war_support = 0.10
		add_command_power = 25
		log = "[GetDateText]: [Root.GetName]: DOP Guangdong entered the southwest war"
	}
}

DOP_GNG_spawn_japanese_reinforcements = {
	if = {
		limit = { NOT = { has_country_flag = DOP_GNG_flow_japanese_reinforcements_arrived } }
		set_country_flag = DOP_GNG_flow_japanese_reinforcements_arrived
		JAP = {
			if = {
				limit = { NOT = { is_in_faction_with = CHI } }
				add_to_faction = CHI
			}
			if = {
				limit = { NOT = { is_in_faction_with = GNG } }
				add_to_faction = GNG
			}
			if = {
				limit = { NOT = { has_war_with = YUN } }
				add_to_war = {
					targeted_alliance = GNG
					enemy = YUN
					hostility_reason = asked_to_join
					single_target_only = yes
				}
			}
			add_dynamic_modifier = { modifier = DOP_GNG_japanese_reinforcement_momentum days = 120 }
			add_equipment_to_stockpile = { type = infantry_equipment amount = 5000 }
			add_equipment_to_stockpile = { type = support_equipment amount = 750 }
			add_equipment_to_stockpile = { type = motorized_equipment amount = 500 }
			add_war_support = 0.10
			add_command_power = 50
			country_event = { id = DOP_GNG_flow.16 hours = 12 }
		}
		log = "[GetDateText]: [Root.GetName]: CHI and GNG rejoined the CPS; JAP entered the southwest war and scheduled Guangzhou deployment in twelve hours"
	}
}

DOP_GNG_complete_southwest_war = {
	if = {
		limit = { has_country_flag = DOP_GNG_flow_southwest_war_active }
		GNG = { white_peace = { tag = YUN } }
		CHI = { white_peace = { tag = YUN } }
		JAP = {
			if = {
				limit = { has_country_flag = DOP_JAP_southwest_reinforcements_deployed }
				delete_unit_template_and_units = {
					division_template = "DOP Nanshin Sensha Shidan"
					disband = no
				}
				clr_country_flag = DOP_JAP_southwest_reinforcements_deployed
			}
			if = {
				limit = { has_war_with = YUN }
				white_peace = { tag = YUN }
			}
		}
		DOP_GNG_remove_southwest_war_state_modifiers = yes

		# Long Shengwu dies at Kunming. Reuse TNO's reconstruction government,
		# but never its differently timed Western Insurrection start effect.
		YUN = {
			if = {
				limit = { has_country_flag = DOP_YUN_southwest_war_prepared }
				retire_character = DOP_YUN_long_shengwu
			}
			clr_country_flag = YUN_long_yun_crusade
			load_focus_tree = {
				tree = YUN_post_xinan
				keep_completed = yes
			}
			YUN_southwest_reconstruction_effects = yes
			set_country_flag = DOP_YUN_southwest_reconstruction_installed
		}

		# Rebuild Guangxi as a Guangdong client, then retain the two coastal
		# states explicitly assigned to Guangdong by the author design.
		GNG = {
			transfer_state = 594
			transfer_state = 599
			transfer_state = 1464
			transfer_state = 2390
			transfer_state = 2391
			transfer_state = 2392
			transfer_state = 2393
			transfer_state = 2394
			transfer_state = 2472
			release_autonomy = {
				target = GUX
				autonomy_state = autonomy_reliant
				freedom_level = 0
			}
			transfer_state = 1464
			transfer_state = 2472
			transfer_state = 591
			transfer_state = 2474
			transfer_state = 2475
		}
		if = {
			limit = { country_exists = GUX }
			GUX = {
				load_focus_tree = {
					tree = ZZZ_blank_focus
					keep_completed = no
				}
			}
		}

		# Yunnan survives as the southwest reconstruction government under CHI.
		CHI = {
			set_autonomy = {
				target = YUN
				autonomy_state = autonomy_military_government
				freedom_level = 0
			}
		}

		clr_country_flag = DOP_GNG_flow_southwest_war_active
		clr_country_flag = DOP_GNG_flow_maoming_countdown_started
		clr_country_flag = DOP_GNG_flow_victory_event_pending
		set_country_flag = DOP_GNG_flow_southwest_war_won
		set_variable = { DOP_GNG_content_stage = 3 }
		remove_ideas = DOP_GNG_wartime_triangle
		remove_ideas = DOP_GNG_strangle_guerrillas
		remove_ideas = DOP_GNG_civil_priority
		remove_ideas = DOP_GNG_civil_priority_supply
		remove_ideas = DOP_GNG_military_priority
		remove_ideas = DOP_GNG_military_priority_industry
		JAP = {
			if = {
				limit = { has_dynamic_modifier = DOP_GNG_japanese_reinforcement_momentum }
				remove_dynamic_modifier = { modifier = DOP_GNG_japanese_reinforcement_momentum }
			}
		}
		DOP_GNG_set_status_postwar = yes
		DOP_GNG_set_security_postwar = yes
		set_temp_variable = { GNG_corruption_status_var_temp = 3 }
		GNG_update_corruption = yes
		DOP_GNG_refresh_fiscal_from_debt = yes
		add_stability = 0.08
		load_focus_tree = dop_sonyjapan_reconsturuction_tree
		log = "[GetDateText]: [Root.GetName]: DOP southwest war completed and reconstruction tree loaded"
	}
}

		limit = {
			NOT = { has_country_flag = DOP_GNG_scw_rank_scored }
			has_country_flag = DOP_SCW_initialized
		}

		SCW_piechart_recalculate = yes
		set_temp_variable = { DOP_GNG_scw_opponents_ahead = 0 }

			limit = { check_variable = { var = GNG_percentage_for_bar value = USA_percentage_for_bar compare = less_than } }
			add_to_temp_variable = { DOP_GNG_scw_opponents_ahead = 1 }
		}
		if = {
			limit = { check_variable = { var = GNG_percentage_for_bar value = ITA_percentage_for_bar compare = less_than } }
			add_to_temp_variable = { DOP_GNG_scw_opponents_ahead = 1 }
		}

			limit = { check_variable = { var = GNG_percentage_for_bar value = GER_percentage_for_bar compare = less_than } }

		}

		subtract_from_variable = { DOP_GNG_scw_rank_points_awarded = DOP_GNG_scw_opponents_ahead }
